#include <iostream>
#include <stdexcept>

template <typename T>
class DoublyLinkedList {
private:
    struct Node {
        T data;
        Node* prev;
        Node* next;

        Node(const T& value) : data(value), prev(nullptr), next(nullptr) {}
    };

    Node* head;
    Node* tail;
    int count;

public:
    DoublyLinkedList() : head(nullptr), tail(nullptr), count(0) {}

    ~DoublyLinkedList() {
        clear();
    }

    // Добавление элемента в конец списка
    void push_back(const T& value) {
        Node* newNode = new Node(value);

        if (tail == nullptr) {
            head = tail = newNode;
        }
        else {
            tail->next = newNode;
            newNode->prev = tail;
            tail = newNode;
        }

        count++;
    }

    // Удаление последнего элемента
    void pop_back() {
        if (tail == nullptr)
            throw std::underflow_error("List is empty");

        Node* temp = tail;
        tail = tail->prev;

        if (tail != nullptr) {
            tail->next = nullptr;
        }
        else {
            head = nullptr;
        }

        delete temp;
        count--;
    }

    // Вывод списка в прямом порядке
    void print() const {
        Node* current = head;

        std::cout << "[";

        while (current != nullptr) {
            std::cout << current->data;

            if (current->next != nullptr)
                std::cout << ", ";

            current = current->next;
        }

        std::cout << "]" << std::endl;
    }

    // Вывод списка в обратном порядке
    void printReverse() const {
        Node* current = tail;

        std::cout << "[";

        while (current != nullptr) {
            std::cout << current->data;

            if (current->prev != nullptr)
                std::cout << ", ";

            current = current->prev;
        }

        std::cout << "]" << std::endl;
    }

    // Обмен соседних элементов
    void swapPairs() {
        Node* current = head;

        while (current != nullptr && current->next != nullptr) {
            Node* first = current;
            Node* second = current->next;

            // Сохраняем следующий элемент после пары
            Node* nextPair = second->next;

            // Меняем связи между элементами
            first->next = nextPair;
            second->prev = first->prev;
            second->next = first;
            first->prev = second;

            // Обновляем предыдущий элемент
            if (second->prev != nullptr) {
                second->prev->next = second;
            }
            else {
                head = second;
            }

            // Обновляем следующий элемент
            if (nextPair != nullptr) {
                nextPair->prev = first;
            }
            else {
                tail = first;
            }

            // Переходим к следующей паре
            current = nextPair;
        }
    }

    // Получение размера списка
    int size() const {
        return count;
    }

    // Очистка списка
    void clear() {
        while (head != nullptr) {
            Node* temp = head;
            head = head->next;
            delete temp;
        }

        tail = nullptr;
        count = 0;
    }
};

int main() {
    setlocale(LC_ALL, "ru");

    DoublyLinkedList<int> list;

    // Добавляем элементы
    for (int i = 1; i <= 5; i++) {
        list.push_back(i * 10);
    }

    std::cout << "=== Двусвязный список ===" << std::endl;

    std::cout << "Прямой порядок: ";
    list.print();

    std::cout << "Обратный порядок: ";
    list.printReverse();

    std::cout << "\n=== Метод swapPairs() ===" << std::endl;

    list.swapPairs();

    std::cout << "После обмена соседних элементов: ";
    list.print();

    std::cout << "Обратный порядок после обмена: ";
    list.printReverse();

    std::cout << "\nРазмер списка: " << list.size() << std::endl;

    list.pop_back();

    std::cout << "После pop_back: ";
    list.print();

    return 0;
}

Рисунок 2 - Вывод

**Контрольные вопросы**
1. Чем связный список отличается от динамического массива?

В динамическом массиве элементы хранятся последовательно в памяти, а в связном списке каждый элемент содержит данные и указатели на другие элементы. В массиве быстрый доступ по индексу, а в списке удобнее вставлять и удалять элементы.

2. Какие операции в односвязном списке выполняются за O(1)?
- Добавление элемента в начало списка (push_front).
- Удаление первого элемента (pop_front).
- Получение первого элемента (head).
- Добавление в конец, если хранится указатель tail.
3. Почему удаление из конца односвязного списка требует O(n) времени?

Потому что необходимо пройти весь список от головы до предпоследнего элемента, чтобы найти его и обновить указатель tail. Односвязный список не позволяет двигаться назад.

4. Какие преимущества даёт хранение указателя tail в односвязном списке?

Указатель tail позволяет добавлять элементы в конец списка за O(1), не проходя каждый раз весь список.

5. Чем двусвязный список отличается от односвязного?

В односвязном списке каждый узел содержит указатель только на следующий элемент (next). В двусвязном списке каждый узел содержит два указателя: на следующий (next) и предыдущий (prev). Поэтому по двусвязному списку можно перемещаться в обоих направлениях.

6. Какие операции в двусвязном списке выполняются быстрее, чем в односвязном?

Удаление последнего элемента при наличии указателя tail выполняется за O(1), тогда как в односвязном списке — за O(n). Также проще удалять известный узел, поскольку можно сразу обратиться к предыдущему элементу.

7. Что такое «голова» (head) и «хвост» (tail) списка?

head — указатель на первый элемент списка.

tail — указатель на последний элемент списка.

Если список пустой, оба указателя равны nullptr.

8. Как удалить узел из середины односвязного списка?

Нужно найти предыдущий узел, изменить его указатель next, чтобы он указывал на следующий после удаляемого узел, а затем удалить сам узел.

9. Как удалить узел из середины двусвязного списка? Почему это проще?

Нужно изменить два указателя: next предыдущего узла и prev следующего узла. Затем удалить нужный узел. Это проще, потому что у каждого узла есть указатель на предыдущий элемент, поэтому не нужно искать его отдельно.

10. Что произойдёт, если забыть обновить tail при удалении?

Указатель tail будет указывать на удалённую область памяти. Это может привести к ошибкам программы, обращению к освобождённой памяти и аварийному завершению.

11. Как реализовать реверс односвязного списка?

Нужно пройти по всем узлам и изменить направление каждого указателя next на противоположное. Для этого используются три указателя: prev, current и next. После завершения голова списка становится последним узлом исходного списка.

12. Как обнаружить цикл в связном списке?

Можно использовать алгоритм Флойда («черепаха и заяц»). Два указателя перемещаются по списку: один на один элемент, другой на два. Если они встретятся, значит, в списке есть цикл.

13. В каких случаях стоит использовать список вместо массива?

Когда требуется часто добавлять и удалять элементы, особенно в начале или середине списка, а быстрый доступ по индексу не является важным.

14. Какая дополнительная память требуется для каждого узла списка?

В односвязном списке — один указатель next.

В двусвязном списке — два указателя next и prev.

Кроме того, каждый узел хранит сами данные.

15. Почему доступ по индексу в списке имеет сложность O(n)?

Потому что список не хранит элементы по непрерывным адресам памяти. Чтобы получить элемент с нужным индексом, необходимо последовательно пройти от головы списка до нужного узла.
